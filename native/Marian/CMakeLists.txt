cmake_minimum_required(VERSION 3.20)
project(localay_marian VERSION 1.0 LANGUAGES CXX)
# Packaging tool only. No FetchContent, submodule download or main-app build hook.
find_package(ctranslate2 4.8.3 EXACT CONFIG REQUIRED)
set(SENTENCEPIECE_ROOT "" CACHE PATH "Installed SentencePiece 0.2.1 prefix")
find_path(SENTENCEPIECE_INCLUDE sentencepiece_processor.h HINTS "${SENTENCEPIECE_ROOT}/include" REQUIRED)
find_library(SENTENCEPIECE_LIBRARY NAMES sentencepiece sentencepiece-static HINTS "${SENTENCEPIECE_ROOT}/lib" REQUIRED)
add_library(localay_marian SHARED bridge.cpp)
target_compile_features(localay_marian PRIVATE cxx_std_17)
target_include_directories(localay_marian PRIVATE "${SENTENCEPIECE_INCLUDE}")
target_link_libraries(localay_marian PRIVATE CTranslate2::ctranslate2 "${SENTENCEPIECE_LIBRARY}")
if(MSVC)
  target_compile_options(localay_marian PRIVATE /W4 /EHsc /utf-8)
endif()
